      # Private engine inputs that are not scraped: an index plus one file per
      # month, read through /api/get-drop (OIDC; that route answers only this
      # workflow on main, and only keys matching KEY_RX below). Never blocks
      # the run: on any failure the files are absent, the sources that need
      # them report an error for this run, and every other town ships.
      # The files land in data/manual/, which the site clone (site/) never
      # reads. This log is public, so the step prints counts only.
      - name: Fetch private engine inputs
        continue-on-error: true
        run: |
          set -uo pipefail
          KEY_RX='^manual/[a-z]{2,40}/20[0-9]{2}-(0[1-9]|1[0-2])(\.r[2-9])?\.json$'
          mint() {
            curl -fsS -H "Authorization: bearer $ACTIONS_ID_TOKEN_REQUEST_TOKEN" \
              "$ACTIONS_ID_TOKEN_REQUEST_URL&audience=masspermits-cron" | jq -r .value
          }
          get() {  # get <key> <dest>
            mkdir -p "$(dirname "$2")"
            curl -fsS --retry 2 --retry-delay 5 --max-time 60 \
              -H "Authorization: Bearer $(mint)" -o "$2" \
              "https://masspermits.com/api/get-drop?key=$1"
          }
          if ! get manual/manifest.json data/manual/manifest.json 2>/dev/null \
             || ! jq -e '.schema == 1 and (.files | type == "array")' data/manual/manifest.json > /dev/null 2>&1; then
            rm -f data/manual/manifest.json
            echo "::warning::private input index not fetched; the sources that need it report an error this run"
            exit 0
          fi
          n=0; bad=0
          for key in $(jq -r '.files[].key' data/manual/manifest.json); do
            if [[ ! "$key" =~ $KEY_RX || "$key" == *..* ]]; then
              bad=$((bad+1)); continue
            fi
            if get "$key" "data/${key}" 2>/dev/null; then n=$((n+1)); else bad=$((bad+1)); fi
          done
          echo "private inputs: $n file(s) fetched, $bad failed"
