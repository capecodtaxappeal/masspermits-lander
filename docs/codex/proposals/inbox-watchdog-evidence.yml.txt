# Problem in the inspected source:
# - curl does not fail on HTTP error status.
# - transport or non-JSON failures become verdict=never, alert=false.
# - valid error JSON without alert also takes the quiet exit.
# This can leave a successful job without usable monitoring evidence.
#
# Narrow proposed replacement for the STATUS collection/validation block:
# Preserve the existing schedule, triggers, permissions and genuine-never policy.
# Use the existing mint function and endpoint. Never print raw failed responses.

STATUS=""
if ! STATUS=$(curl -fsS -H "Authorization: Bearer $(mint)" \
  https://masspermits.com/api/inbox-status); then
  STATUS='{"verdict":"unknown","alert":true,"detail":"Inbox monitoring evidence could not be retrieved. Check the existing run and status service."}'
fi
if ! printf '%s' "$STATUS" | jq -e '
  type == "object"
  and (.verdict | type == "string")
  and (.alert | type == "boolean")
  and (.detail | type == "string")
  and (.verdict == "ok" or .verdict == "waiting" or .verdict == "never"
       or .verdict == "stale" or .verdict == "unarmed"
       or .verdict == "backlog" or .verdict == "unknown")
' >/dev/null 2>&1; then
  STATUS='{"verdict":"unknown","alert":true,"detail":"Inbox monitoring evidence has an invalid response shape. Check the status service."}'
fi

# Replace the raw JSON print with a validated public summary. Additional numeric
# fields may be emitted only after numeric validation. Do not log raw DETAIL.
printf '%s' "$STATUS" | jq '{verdict, alert}'
